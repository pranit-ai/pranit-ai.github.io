MY WORKING PRINCIPLE 
def deliver_solution(business_problem):
    outcome = understand(business_problem)
    requirements = define_requirements(outcome)
    context = assess_data_and_systems(requirements)

    design = design_architecture(requirements, context)
    solution = build_and_integrate(design)

    while not success_criteria_met(solution):
        results = evaluate(solution, metrics=outcome.kpis)
        solution = iterate(solution, results, user_feedback=True)

    return deploy_and_monitor(solution)
